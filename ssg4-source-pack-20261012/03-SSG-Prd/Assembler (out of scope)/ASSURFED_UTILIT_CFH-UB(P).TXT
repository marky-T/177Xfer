          SX,H1     X11,P          . SAVE RETURN LINK

          LR,U      R1,28          . READ 28 WORDS FORWARD

          ER        IOW$           . READ BLOCK 
        L       A0,STAT 
          TE,016    A0,4
        JNZ     A0,IOERR
         SLJ       CHECPT     
        LA      A0,STAT 
        JNZ     A0,IOERR
          LXM       A0,A2          . GET LABEL ROUTINE NAME 
          SSL       A2,18          . SHIFT TO GET LABEL     
          LA,U      A3,LABEL            . LABEL AREA  
          TNE,H1    A2,0,A3        . TEST BLOCKTYPE   
          J         LREIT          . CORRECT BLOCKTYPE
          TE,U      A2,'EOF'       . ATTEMPTED CLOSE FILE TEST    
          LMJ       X11,LERR$10    . BLOCKTYPE ERROR  
          LA,H1     A1,0,A3        . GET BLOCKTYPE    
          TE,U      A1,'EOR'       . END OF REEL BLOCK TEST 
          LMJ       X11,LERR$10    . BLOCKTYPE ERROR  
          J         LREEXIT        . ACCEPTABLE BLOCK BUT NO LABEL RPOC 
LREIT     TNE,U     A2,'HDR'       . DETERMINE BLOCKTYPE FOR PROCESSING 
          J         $+3            . HDR LABEL,MUST GET INFO
          TNZ       FAST           . FASTRAND EOF MUST CHECK INFO 
          J         LRECONT        . NO CHECKING NECESSARY  
          LA        A1,5,A3        . GET REEL NUMBER  
          SA        A1,REEL        . SAVE 
          LA,H1     A1,1,A3        . GET BLOCKSIZE    
          SA        A1,MINBLK      . SAVE 
          AN,U      A1,1           .
          TG        A1,MAXBLK      . IS BLKSIZE TOO LARGE   
          LMJ       X11,LERR$11    . ERROR
          LA,H2     A1,1,A3        . GET RECORD SIZE  
          SA        A1,MINREC      . SAVE 
          AN,U      A1,1           .
          TG        A1,MAXREC      . IS RECORD SIZE TOO LARGE     
          LMJ       X11,LERR$12    . ERROR
          TZ        INOUT              .  
          TE,U      A2,'HDR'           . SKIP IF INOUT HEADR BLOCK
          J         $+2

          SA        A1,BLKCT       . INSERT IN FCT    
          LA        A1,3,A3        . GET RECORD COUNT 
          SA        A1,RECCT       . INSERT IN FCT    
          L         A1,6,A3            . FASTRAND END OF FILE     
          S         A1,FEOF            .  
LRECONT   TZ        OLAB           . OMITTED LABEL TEST     
          J         LROMIT    
          L         A1,USEL          . SAVE USEL
          S         A1,6,X10         .    
          L,U       A1,14          . STANDARD LABEL SIZE    
          SA        A1,USESIZ      .
          AA        A1,A3          . ADD BEGN BUFFER ADDRESS
          SA        A1,USELOC      .
          J         0,A0           . JUMP TO APPROPRIATE LABEL PROCEDURE
LREEOF    LABROUT   BSEFI,SIP,ASEFI . CALL ENDING FILE LABEL ROUTINES   
          J         LRENO          .
LREBOR    LABROUT   BSBRI,SIP,ASBRI . CALL BEGN REEL ROUTINES     
          J         LRENO          .

          J         LRENO          .
LREBIO    LABROUT   BSBFIO,SIP,ASBFIO   .  BEGN FILE INPUT-OUTPUT 
LRENO     L         A0,6,X10  
          S         A0,USEL            .  
LROMIT    TZ        REV 
          TE,U      A2,'HDR'       . MOVE OVER EOF MARK ON REV AFTER EOF
          TZ        FAST           . NO EOFMARK ON FASTRAND 
          J         LREEXIT        .
          LA,U      A1,MF$         . MOVE FORWARD COD(
          TZ        REV            .
          LA,U      A1,MB$         . MOVE BACKWARDS   
          SA        A1,FUNC        .
          LA        A0,X10    
          ER        IOW$           .
LREEXIT   LX,H1     X11,P          .
          LA        A0,X10         . RELOAD I/O PKT ADDRESS 
          J         0,X11          .
