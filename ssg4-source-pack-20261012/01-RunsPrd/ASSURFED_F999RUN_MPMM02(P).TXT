@LAB020: MSG 020. UNLOAD DES PARAMETRES SW513-SW514 
@ . 
@ADD SYS$*RUNS.REPRISE  
LAB020  
@ . 
@SETC      0/S2 
@DELETE,C MPMM02*IPARAMETRES.   
@CAT,P    MPMM02*IPARAMETRES.,F/100/POS/4000
@ASG,A    MPMM02*IPARAMETRES.   
@XQT,E SYS$LIB$*RSA.RSAC-COIVP  
BEGIN THREAD FOR UDSSRC READ;   
USE DEFAULT QUALIFIER AFLINC;   
UNLOAD TO FILE "MPMM02*IPARAMETRES."
  SELECT 'PARAM ', VALPARAM, N_PARAM1, D_PARAM, TIMEPARAM   
  FROM SW513 A ,SW514 B 
  WHERE A.IDRAP1     = 'MPMM02'        AND  
        A.C_FLAG     = '2'             AND  
        A.IDRAP1     = B.IDRAP1        AND  
        A.D_PARAM    = B.D_PARAM       AND  
        A.TIMEPARAM  = B.TIMEPARAM  
  ORDER BY D_PARAM DESC, TIMEPARAM DESC, N_PARAM1 ASC   
  FORMAT EXTERNAL;  
END THREAD; 
EXIT;   
@FREE      MPMM02*IPARAMETRES.  
@TEST      TE/0/S2  
@JUMP      ERRC 
@JUMP      OK   
@ERRC: MSG ** PROBLEME DANS L'UNLOAD ** 
@ABORT  
@OK:   MSG ** UNLOAD OK **  
@ERS         TPF$.  
@TIME
