@LAB160: MSG 160. LISTE DES DEPENSES PAR SINISTRE.           RSA
@ .      =======================================================
@ . Liste des dépenses entre 1/1 et 31/12 année écoulée 
@ADD       SYS$*RUNS.REPRISE
LAB160  
@ . 
@SETC      0/S2 
@DELETE,C  SCTA02*IMP-RSA.  
@ASG,UP    SCTA02*IMP-RSA.,F///4000 
@XQT,E SYS$LIB$*RSA.RSAC-COIVP  
BEGIN THREAD FOR UDSSRC READ;   
USE DEFAULT QUALIFIER AFLINC;   
UNLOAD TO FILE "SCTA02*IMP-RSA" 
    
SELECT no_sin, sum(md_cpt)  
  FROM SCD201_OPR_FNC     SOF,  
       SCD202_LGN_MON     SLM,  
       SCD108_CAT_CPT_COO SCCC  
 WHERE SOF.CD_ETA         = 'DEBI'  
   AND (
        (   
         month(curdate()) = 12  
         AND
         SOF.DT_ETA_TRI between  year(curdate()) * 10000
                            and (year(curdate()) + 1) * 10000   
        )   
        OR  
        (   
         month(curdate()) = 1   
         AND
         SOF.DT_ETA_TRI between (year(curdate()) -1 ) * 10000   
                            and  year(curdate()) * 10000
        )   
       )
   AND SOF.FL_CPN_GEN     = 'Y' 
   AND SLM.NO_OPR_FNC     = SOF.NO_OPR_FNC  
   AND SLM.CD_CAT_CPT     = SCCC.CD_CAT_CPT 
   AND SCCC.DT_DEB       <= SOF.DT_CPT  
   AND SCCC.DT_FIN       >= SOF.DT_CPT  
   AND SCCC.FL_CUT        = 'Y' 
 GROUP BY no_sin
 ORDER BY no_sin

FORMAT EXTERNAL;
EXIT;   
@FREE      SCTA02*IMP-RSA.  
@TEST      TE/0/S2  
@JUMP      ERRC1
@JUMP      OK1  
@ERRC1: MSG ** PROBLEME DANS L'UNLOAD **
@ABORT  
@OK1:   MSG ** UNLOAD OK ** 
@TIME
