@use fl.,LRSM14*IDRLR401.   
@XQT,E    SYS$LIB$*RSA.RSAC-COIVP   
BEGIN THREAD FOR UDSSRC READ;   
USE DEFAULT QUALIFIER AFLINC       ;
 unload to file fl  
 select n_mat,n_dcl,c_branche,policec,c_num_avn,
  n_rsq_mat from lr401 a , pr_lr4012 c where
      (a.d_fin = 0 or a.d_fin > 20050100) and c.c_gnr_dcl = 'D' and 
  exists (select * from lr401 b 
          where a.n_mat = b.n_mat and c.c_gnr_dcl = 'D' and 
   a.n_rsq_mat = b.n_rsq_mat and a.n_dcl = b.n_dcl and  
   not a.c_branche = b.c_branche and (b.d_fin = 0 or b.d_fin > 20050100) )  
  and c.c_branche in ('100','107','109') and c.n_mat = a.n_mat and  
  a.n_rsq_mat = c.n_rsq_mat and a.c_branche = c.c_branche and   
  a.policec = c.policec and a.c_typ_avn = c.c_typ_avn and   
  a.c_num_avn = c.c_num_avn and a.d_effet = c.d_effet and   
  a.c_gnr_dcl = c.c_gnr_dcl and a.c_cat_emr = c.c_cat_emr and   
  a.n_cmm_pta = c.n_cmm_pta and a.n_dcl = c.n_dcl   
  order by n_mat,c_branche,policec,c_num_avn
  format external;  
END THREAD; 
EXIT;

            C.N_TRT_AUM,C.IDTXT,C.C_ILE_RED,C.D_DEB,C.N_PRG 
        FROM
            LR429 A 
            LEFT JOIN LR407 C   
            ON A.N_MAT = C.N_MAT
        WHERE   
            A.C_STA_TRT = 'ATP' 
            AND 
            EXISTS (SELECT * FROM AL027 B   
                    WHERE A.N_MAT = B.MATRICC AND B.C_SUSPEND <> '32')
