LAB110  
@ .

@DELETE,c  RISH02*IUNLOADRIV.   
@cat,p     RISH02*IUNLOADRIV.,f/0/POS/4000  
@asg,a     RISH02*IUNLOADRIV.   
@use       f-unload1,RISH02*IUNLOADRIV  
@ . 
@xqt,e     SYS$LIB$*RSA.RSAC-COIVP  
BEGIN THREAD FOR UDSSRC READ UDSMSG;
USE DEFAULT QUALIFIER AFLINC;   
    
unload to file "F-UNLOAD1"  
   select   
         Substring (Rtrim (Ltrim (Rmb.LB_CMT_001)),1,10),   
         Rmb.MT_RMB,
         Rtrim (Ltrim (Rmb.LB_CMT_002)),
         Rtrim (Ltrim (Rmb.LB_CMT_003)),
         Rtrim(ltrim(Don_Ecp_Trs.TL_ECP_001)),  
         Rtrim(ltrim(Don_Ecp_Trs.TL_ECP_002)),  
         Rtrim(ltrim(Don_Ecp_Trs.TL_ECP_003)),  
         Rtrim(ltrim(Don_Ecp_Trs.TL_ECP_004)),  
         Rmb.LB_CMT_001 
    
    From Tmp_Seq_Rdm, Rmb, Don_Ecp_Trs  
    
      Where 
          Don_Ecp_Trs.CD_TYP_TRS     = 'A'  
      AND REPLACE(STR(Don_Ecp_Trs.NO_TRS,10),' ','0') = 
          Substring (Rtrim (Ltrim (Rmb.LB_CMT_001)),1,10)   
      AND Don_Ecp_Trs.CD_TYP_CTE_TRS = 'I'  
      AND Don_Ecp_Trs.CD_OBJ_RCV     IN (' ', 'RI') 
      AND Don_Ecp_Trs.NO_BCH         = ' '  
      AND Don_Ecp_Trs.NO_POL         = 0
      AND Don_Ecp_Trs.CD_TYP_ECP     = 830  
      AND Rmb.MR_DAT_MAJ Between
                Don_Ecp_Trs.DT_DEB And Don_Ecp_Trs.DT_FIN   
      AND Tmp_seq_rdm.no_seq     = Rmb.no_Lot   
      AND Tmp_seq_rdm.mr_run_maj = 'RISH02' 
      AND Tmp_seq_rdm.mr_prg_maj = 'RIB010' 
      AND Tmp_seq_rdm.nm_fch_log = 'VRI'
    
    Order By Rmb.LB_CMT_001 
    
    format internal;
    ;   
exit;
