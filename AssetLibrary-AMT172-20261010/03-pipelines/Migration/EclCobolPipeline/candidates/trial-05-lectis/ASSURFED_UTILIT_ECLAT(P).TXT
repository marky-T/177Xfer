      * 
       PROCEDURE DIVISION.  
      **********************
       000-MAIN.
      **********************
           PERFORM 010-OUVERTURE
           PERFORM 020-TRT-PRINC UNTIL (EOF-CHER = "1" AND  
                                       EOF-FILE = "1")  
           IF EOF-FILE = "0"
              PERFORM LEC-FILE  
      *       PERFORM 030-TRT-SEC UNTIL (EOF-FILE = "1")
           END-IF   
           PERFORM 090-CLOTURE  
           STOP RUN.
      * 
      ******FONCTIONS*******
      **********************
       LEC-CHER.
           READ 1 AT END MOVE "1" TO EOF-CHER  *> SSG-STUB
           END-READ 
           IF EOF-CHER = "0"
              ADD 1 TO CPT-CHER 
              MOVE KEY1 to S-KEY1  *> SSG-STUB
           ELSE 
              MOVE HIGH-VALUE TO SEQ-KEY
           END-IF   
           .
       LEC-FILE.
           READ 1 AT END MOVE "1" TO EOF-FILE    *> SSG-STUB
           END-READ 
           IF EOF-FILE = "0"
              ADD 1 TO CPT-FILE 
              MOVE KEY-T1 to S-KEY-T1  *> SSG-STUB
           ELSE 
              MOVE HIGH-VALUE TO SEQ-KEY-T  
           END-IF   
           .
      * 
       010-OUVERTURE.   
      **********************
           OPEN INPUT 1 1  *> SSG-STUB
                OUTPUT FARCHIVE FENCOURS
           ACCEPT DAT-DAT FROM DATE 
           PERFORM LEC-CHER 
           PERFORM LEC-FILE.
      * 
       020-TRT-PRINC.   
      **********************
           IF SEQ-KEY = SEQ-KEY-T   
                MOVE "1" TO NOM-TAB  *> SSG-STUB
                IF DAT-AA < 28  
                   MOVE 20 TO DAT-SS
                ELSE
                   MOVE 19 TO DAT-SS
                END-IF  
                MOVE SEQ-KEY TO CLES-TAB
                MOVE ENR-TAB TO ENR 
                WRITE ENR-ARC   
                ADD 1 TO CPT-ARC
                PERFORM LEC-CHER
                PERFORM LEC-FILE
           ELSE 
             IF SEQ-KEY > SEQ-KEY-T 
                MOVE ENR-TAB TO ENR-ENC 
                WRITE ENR-ENC   
                ADD 1 TO CPT-ENC
                PERFORM LEC-FILE
             ELSE   
      *         ERREUR  
                PERFORM LEC-CHER
             END-IF 
           END-IF.  
      * 
       030-TRT-SEC. 
      **********************
           MOVE ENR-TAB TO ENR-ENC  
           WRITE ENR-ENC
           ADD 1 TO CPT-ENC 
           PERFORM LEC-FILE.
      * 
       090-CLOTURE. 
      **********************
           CLOSE 1 1 FARCHIVE FENCOURS *> SSG-STUB
           MOVE CPT-ENC TO CPT-ENC-ED   
           MOVE CPT-ARC TO CPT-ARC-ED   
           MOVE CPT-CHER TO CPT-CHER-ED 
           MOVE CPT-FILE TO CPT-FILE-ED 
           DISPLAY   '  LES TOTAUX DE CONTROLE:' UPON PRINER
           DISPLAY   '  ***********************' UPON PRINER
           DISPLAY             ' Nombre de records lus dans le fichier' 
           ' 1 : ' CPT-CHER-ED UPON PRINER *> SSG-STUB
           DISPLAY             ' Nombre de records lus dans le fichier' 
           ' 1 : ' CPT-FILE-ED UPON PRINER   *> SSG-STUB
           DISPLAY           ' Nombre de records {crit dans le fichier' 
           ' 1 : ' CPT-ARC-ED UPON PRINER *> SSG-STUB
           DISPLAY           ' Nombre de records {crit dans le fichier' 
           ' 1 : ' CPT-ENC-ED UPON PRINER.    *> SSG-STUB
