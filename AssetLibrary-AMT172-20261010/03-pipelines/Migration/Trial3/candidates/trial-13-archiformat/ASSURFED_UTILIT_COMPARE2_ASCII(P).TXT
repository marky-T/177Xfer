       IDENTIFICATION DIVISION. 
       PROGRAM-ID.    COMPARE.  
       AUTHOR.        HANSE.
       ENVIRONMENT DIVISION.
       CONFIGURATION SECTION.   
       SOURCE-COMPUTER. SPERRY-1100.
       OBJECT-COMPUTER. SPERRY-1100.
       INPUT-OUTPUT SECTION.
       FILE-CONTROL.
           SELECT IN1 ASSIGN TO DISC FILEIN-1.  *> SSG-STUB [INPUT,1,1,1]
           SELECT IN2 ASSIGN TO DISC 80.  *> SSG-STUB [INPUT,1,1,2]
           SELECT LISTING ASSIGN TO PRINTER.
       DATA DIVISION.   
       FILE SECTION.
       FD  IN1  
           LABEL  RECORD IS OMITTED 
           LABEL  RECORD IS STANDARD
           BLOCK  CONTAINS 80 RECORDS  *> SSG-STUB [INPUT,1,1,4]
           RECORD CONTAINS 1 CHARACTERS  *> SSG-STUB [INPUT,1,1,3]
           DATA RECORD IS ENR-IN1.  
       01  ENR-IN1   PIC X(1).  *> SSG-STUB [INPUT,1,1,3]
       FD  IN2  
           LABEL  RECORD IS OMITTED 
           LABEL  RECORD IS STANDARD
           BLOCK  CONTAINS 80 RECORDS  *> SSG-STUB [INPUT,1,1,4]
           RECORD CONTAINS 1 CHARACTERS  *> SSG-STUB [INPUT,1,1,3]
           DATA RECORD IS ENR-IN2.  
       01  ENR-IN2   PIC X(1).  *> SSG-STUB [INPUT,1,1,3]
       FD  LISTING  
           LABEL RECORD IS OMITTED  
           BLOCK CONTAINS 1 RECORDS 
           RECORD CONTAINS 132 CHARACTERS   
           DATA RECORD IS LIGNE.
       01  LIGNE PIC X(132).
       WORKING-STORAGE SECTION. 
    
       01  L-NOLIGNE.   
           05 FILLER PIC X(50) VALUE
           '----+----1----+----2----+----3----+----4----+----5'.
           05 FILLER PIC X(50) VALUE
           '----+----6----+----7----+----8----+----9----+----1'.

       01  WS-IN1.  
           02 TAB-IN1 OCCURS 1 INDEXED BY XT1 PIC X(100).  *> SSG-STUB [*A]
       01  WS-IN11 REDEFINES WS-IN1.  *> SSG-STUB [*I]
           02 INDIC11      PIC X(1).  *> SSG-STUB [*I], [INDIC,1,I,2]
       01  WS-EXCL1-1 REDEFINES WS-IN1.  *> SSG-STUB [*I]
           02 FILLER                PIC X(-1).  *> SSG-STUB [*L1]
           02 EXCL1-1              PIC X(1).  *> SSG-STUB [*I], [EXCLUDE,1,I,2]
           02 FILLER                PIC X(1).  *> SSG-STUB [*L2]
       01  WS-IN2.  
           02 TAB-IN2 OCCURS 1 INDEXED BY XT2 PIC X(100).  *> SSG-STUB [*A]
       01  WS-IN21 REDEFINES WS-IN2.  *> SSG-STUB [*I]
           02 INDIC21      PIC X(1).  *> SSG-STUB [*I], [INDIC,1,I,2]
       01  WS-EXCL2-1 REDEFINES WS-IN2.  *> SSG-STUB [*I]
           02 FILLER                PIC X(-1).  *> SSG-STUB [*L1]
           02 EXCL2-1              PIC X(1).  *> SSG-STUB [*I], [EXCLUDE,1,I,2]
           02 FILLER                PIC X(1).  *> SSG-STUB [*L2]
       01  WS-INDIC1.   
           02 IND11        PIC X(1).  *> SSG-STUB [*I], [INDIC,1,I,2]
           02 FILLER          PIC X(4). 
       01  WS-INDIC2.   
           02 IND21        PIC X(1).  *> SSG-STUB [*I], [INDIC,1,I,2]
           02 FILLER          PIC X(4). 
       01  LIGNE-NO-REC.
         03  FILLER   PIC X(20) VALUE 'ENREGISTREMENT NO   '.   
         03  ZCT1     PIC 9(10) VALUE ZEROES.   
         03  FILLER   PIC X(4) VALUE ' (FILEIN-1) '.  *> SSG-STUB [INPUT,1,1,1], [*LONGUEUR]
         03  FILLER   PIC X(3) VALUE ' / '. 
         03  ZCT2     PIC 9(10) VALUE ZEROES.   
         03  FILLER   PIC X(4) VALUE ' (80) '.  *> SSG-STUB [INPUT,1,1,2], [*LONGUEUR]
         03  FILLER   PIC X(9) VALUE 'NO DIFF: '.   
         03  ZCT3     PIC 9(6) VALUE ZEROES.
       01  CT1     PIC 9(10) COMP VALUE ZEROES. 
       01  CT2     PIC 9(10) COMP VALUE ZEROES. 
       01  SW-ANO  PIC 9          VALUE 0.  
       01  DIFFERENCE         PIC 9            VALUE 0. 
       01  NB-TOT-DIFFERENCE  PIC 9(6)         VALUE 0. 
       01  WS-COMPTEUR-PRNT   PIC 9(6)         VALUE 0. 
       PROCEDURE DIVISION.  
       DEBUT.   
           OPEN INPUT IN1 IN2   
                OUTPUT LISTING. 
           MOVE SPACES TO LIGNE.
           WRITE LIGNE AFTER PAGE.  
       LECTURE. 
           PERFORM LECTURE-IN1 THRU LECTURE-EX. 
           PERFORM LECTURE-IN2 THRU LECTURE-EX. 
       TEST.
           IF WS-INDIC1 = WS-INDIC2  GO TO EGALITE. 
           IF WS-INDIC1 > WS-INDIC2  GO TO IN2-SEUL.
           IF WS-INDIC1 < WS-INDIC2  GO TO IN1-SEUL.
       EGALITE. 
           IF WS-INDIC1 = HIGH-VALUES GO TO FINI.   
           GO TO LECTURE
           MOVE 0 TO SW-ANO.
           SET XT1 XT2 TO 1.
       BCL1.
           IF XT1 > 1  GO TO IMP-EGAL.  *> SSG-STUB [*A]
           IF XT2 > 1  GO TO IMP-EGAL.  *> SSG-STUB [*A]
           IF TAB-IN1 (XT1) = TAB-IN2 (XT2)  GO TO RBCL1.   
           MOVE 1 TO DIFFERENCE.
           MOVE 1 TO SW-ANO.
       RBCL1.   
           SET XT1 XT2 UP BY 1. 
           GO TO BCL1.  
       IMP-EGAL.
           IF SW-ANO = 0   GO TO LECTURE.   
           ADD  1 TO WS-COMPTEUR-PRNT.  
           IF NOT WS-COMPTEUR-PRNT > 0  *> SSG-STUB [*AIMPRIMER]
              GO TO IMPRESSION. 
           GO TO LECTURE.   
       IMPRESSION.  
           MOVE CT1 TO ZCT1.
           MOVE CT2 TO ZCT2.
           DISPLAY ' ' UPON PRINTER 
           DISPLAY LIGNE-NO-REC                   UPON PRINTER  
           DISPLAY L-NOLIGNE                      UPON PRINTER. 
           SET XT1 TO 1.
           SET XT2 TO 1.
       BCL-IMP. 
           IF XT2 > 1   GO TO LECTURE.  *> SSG-STUB [*A]
           DISPLAY TAB-IN1 (XT1) UPON PRINTER.  
           DISPLAY TAB-IN2 (XT2) UPON PRINTER.  
           SET XT1 UP BY 1  
           SET XT2 UP BY 1. 
           GO TO BCL-IMP.   
       IN1-SEUL.
           PERFORM LECTURE-IN1 THRU LECTURE-EX. 
           GO TO TEST.  
           DISPLAY '< ENREG. NO : ' CT1 UPON PRINTER.   
           SET XT1 TO 1.
       BCL-IMP-IN1. 
           IF XT1 > 1   PERFORM LECTURE-IN1 THRU LECTURE-EX  *> SSG-STUB [*A]
                           GO TO TEST.  
           DISPLAY TAB-IN1 (XT1) UPON PRINTER.  
           SET XT1 UP BY 1. 
           GO TO BCL-IMP-IN1.   
           PERFORM LECTURE-IN1 THRU LECTURE-EX. 
           GO TO TEST.  
       IN2-SEUL.
           PERFORM LECTURE-IN2 THRU LECTURE-EX. 
           GO TO TEST.  
           DISPLAY '> ENREG. NO :            - ' CT2 UPON PRINTER.  
           SET XT2 TO 1.
       BCL-IMP-IN2. 
           IF XT2 > 1   PERFORM LECTURE-IN2 THRU LECTURE-EX  *> SSG-STUB [*A]
                           GO TO TEST.  
           DISPLAY TAB-IN2 (XT2) UPON PRINTER.  
           SET XT2 UP BY 1. 
           GO TO BCL-IMP-IN2.   
           PERFORM LECTURE-IN2 THRU LECTURE-EX. 
           GO TO TEST.  
       LECTURE-IN1. 
           ADD DIFFERENCE TO NB-TOT-DIFFERENCE. 
           MOVE NB-TOT-DIFFERENCE TO ZCT3.  
           ADD 1 TO ZCT3.   
           MOVE 0 TO DIFFERENCE.
           READ IN1 INTO WS-IN1 AT END  
                                MOVE HIGH-VALUES TO WS-INDIC1   
                                GO TO LECTURE-EX.   
           PERFORM EXCLUDE1 THRU EXCLUDE1-EXIT. 
           ADD 1 TO CT1.
           MOVE INDIC11 TO IND11.  *> SSG-STUB [*I], [*I]
           GO TO LECTURE-EX.
       LECTURE-IN2. 
           MOVE 0 TO DIFFERENCE.
           READ IN2 INTO WS-IN2 AT END  
                                MOVE HIGH-VALUES TO WS-INDIC2   
                                GO TO LECTURE-EX.   
           PERFORM EXCLUDE2 THRU EXCLUDE2-EXIT. 
           ADD 1 TO CT2.
           MOVE INDIC21 TO IND21.  *> SSG-STUB [*I], [*I]
       LECTURE-EX.  
           EXIT.
       EXCLUDE1.
           MOVE ALL '*' TO EXCL1-1.  *> SSG-STUB [*I]
       EXCLUDE1-EXIT.   
           EXIT.
    
       EXCLUDE2.
           MOVE ALL '*' TO EXCL2-1.  *> SSG-STUB [*I]
       EXCLUDE2-EXIT.   
           EXIT.
    
    
       FINI.
           DISPLAY '******************************' 
                    UPON PRINTER.   
           DISPLAY 'NOMBRE ENREGISTREMENTS INPUT FICHIER 1 = ' CT1  
                    UPON PRINTER.   
           DISPLAY 'NOMBRE ENREGISTREMENTS INPUT FICHIER 2 = ' CT2  
                    UPON PRINTER.   
           DISPLAY 'NOMBRE ENREGISTREMENTS DIFFERENTS = '   
                    NB-TOT-DIFFERENCE              UPON PRINTER.
           DISPLAY '******************************' UPON PRINTER.   
           DISPLAY '**********************************************' 
                    UPON PRINTER.   
           IF NB-TOT-DIFFERENCE > 0  *> SSG-STUB [*AIMPRIMER]
                 DISPLAY '* TOUTES LES DIFFERENCES NE SONT PAS LISTEES *'   
                 UPON PRINTER   
           ELSE 
                 DISPLAY '*    TOUTES LES DIFFERENCES SONT LISTEES     *'   
                 UPON PRINTER.  
           DISPLAY '**********************************************' 
                    UPON PRINTER.   
           CLOSE IN1 IN2 LISTING.   
           STOP RUN.
