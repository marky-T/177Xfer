      *------------*
           MOVE "0" TO WS-ESQL-FL-THREAD

           EXEC SQL BEGIN THREAD READ END-EXEC  
    
           IF  SQLCODE NOT = 0  
                MOVE "100- BEGIN THREAD" TO WS-ESQL-TRACE

                CALL "EREABT$"

           EXEC SQL  USE DEFAULT QUALIFIER AFLINC END-EXEC  
           EXEC SQL  USE DEFAULT VERSION PRODUCTION  END-EXEC   
           IF  SQLCODE NOT = 0  
                PERFORM 999-ESQL-ERR
