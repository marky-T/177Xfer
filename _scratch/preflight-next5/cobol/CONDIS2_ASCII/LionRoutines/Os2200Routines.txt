var

{*
change mainframe style of names to amt style
        filler: alpha 12 value '(PRODUCTION)'
        filler: alpha 18 value 'CAT/CURYRFL ON DBB'
        filler: alpha 1 value '.'

so the value '(PRODUCTION)CAT/CURYRFL ON DBB.'
is changed to 'CAT/CURYRFL.DAT'
*}

routine main (s_filenamenew: string, b_print : boolean optional, s_filenameold : string optional, s_printid: string optional): string
var
    i_pos : integer
    s_tmp : string
begin_routine
    // Check if the new filename is already fully qualified
    if copy(s_filenamenew, 1, 2) = '\\' or copy(s_filenamenew, 2, 1) = ':'
        // filenamenew is already complete
        result := s_filenamenew
        exit
    endif
    if s_filenamenew = ''
      s_filenamenew := s_filenameold
    endif

    // Remove possible username
    if copy(s_filenamenew, 1, 1) = '('
        i_pos         := 0
        s_tmp         := unstring (s_filenamenew, i_pos, ')')
        s_filenamenew := copy(s_filenamenew, i_pos)
    endif

    // Remove possible " ON <pack>"
    i_pos := 0
    s_tmp := unstring(s_filenamenew, i_pos, ' ON ')
    if i_pos < length(s_filenamenew)
        s_filenamenew := copy(s_filenamenew, 1, i_pos - 4)
    endif

    // Remove possible dot at the end
    i_pos := length(s_filenamenew)
    if copy(s_filenamenew, i_pos, 1) = '.'
        s_filenamenew := copy(s_filenamenew, 1, i_pos - 1)
    endif

    // add file extension 
    if b_print
        result := s_filenamenew & '.PRN'
    else
        result := s_filenamenew & '.DAT'
    endif
end_routine
