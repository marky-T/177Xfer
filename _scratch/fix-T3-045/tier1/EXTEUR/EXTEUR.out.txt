      *=================================================================*   
       030-INIT-DB. 
      *------------*
           MOVE "0" TO WS-ESQL-FL-THREAD
           EXEC SQL BEGIN THREAD READ END-EXEC  
    
           IF  SQLCODE NOT = 0  
                MOVE "100- BEGIN THREAD" TO WS-ESQL-TRACE   
                PERFORM 999-ESQL-ERR
                PERFORM 090-CLOTURE THRU 090-EXIT   
                CALL "EREABT$"  
           END-IF   
           MOVE "1" TO WS-ESQL-FL-THREAD
           EXEC SQL  USE DEFAULT QUALIFIER AFLINC END-EXEC  
           EXEC SQL  USE DEFAULT VERSION PRODUCTION  END-EXEC   
           IF  SQLCODE NOT = 0  
                PERFORM 999-ESQL-ERR
                PERFORM 090-CLOTURE THRU 090-EXIT   
                CALL "EREABT$"  
           END-IF   
           .
       030-EXIT.  EXIT. 
      *---------*   
    
       035-RECHERCHE-PHASE. 
      *-------------------* 
           MOVE 0 TO C-PHA-EUR. 
           EXEC SQL 
                 SELECT 
                       C_PHA_EUR
                    INTO :C-PHA-EUR 
                    FROM
                       EU001
           END-EXEC 
    
           IF C-PHA-EUR = '100' OR = '200'  
              MOVE 'BEF' TO WS-DEVISE-2200  
           END-IF   
           IF C-PHA-EUR = '300' OR = '350' OR = '400'   
              MOVE 'EUR' TO WS-DEVISE-2200  
           END-IF   
    
           DISPLAY 'PHASE ACTUELLE : ' C-PHA-EUR '/' WS-DEVISE-2200 
           .
       035-EXIT.  EXIT.

            READ INA AT END MOVE 1 TO SW-END.

            ADD 1 TO WS-NB-REC-LUS

                 REC-FILE-OUTA

               IF DEVISE-INA = '1'

            END-IF  
    
            IF WS-DEVISE-2200 = 'EUR'   
               IF DEVISE-INA NOT = '1' AND NOT =   
                                   '2' 
                  DISPLAY '*---------------------------------*' 
                  DISPLAY '*     GRAVE ERREUR DE DEVISE      *' 
                  DISPLAY '*        ABORT DU PROGRAMME       *' 
                  DISPLAY '*        CONTACTER J-Y NAUW       *' 
                  DISPLAY '*       APPELER LES POMPIERS      *' 
                  DISPLAY '*---------------------------------*'

               END-IF

       999-ESQL-ERR.
      *-----------------------------------------------------------------*   
      * ESQL ERROR                                                      *   
      *-----------------------------------------------------------------*   
           IF  SQLCODE = -6010  
                MOVE CS-ESQL-NO-MSG-DEADLOCK TO MSGERR-NO   
                MOVE '1'                     TO MSGERR-CD-RLG   
                MOVE WS-ESQL-TRACE           TO MSGERR-VAR-001  
                CALL 'MSGERR' USING MSGERR  
                DISPLAY MSGERR-LB (1:60) UPON CONSOLE   
                CALL 'BLOACE' USING WS-ESQL-TRACE   
                ADD  1 TO WS-ESQL-CT-DEADLOCK   
                MOVE 1 TO WS-ESQL-FL-DEADLOCK   
                IF  WS-ESQL-CT-DEADLOCK < 6 
                     PERFORM 090-CLOTURE THRU 090-EXIT  
    
                     GO TO 000-MAIN 
                END-IF  
           END-IF   
           .
           PERFORM 999-ESQL-GET-ERR 
           IF WS-ESQL-FL-THREAD = '1'   
                EXEC SQL ROLLBACK WORK END-EXEC 
      *         EXEC SQL END THREAD    END-EXEC 
      *         MOVE '0' TO WS-ESQL-FL-THREAD   
           END-IF   
           .
       999-ESQL-ERR-NO-DISP.
      *-----------------------------------------------------------------*   
      * ESQL ERROR                                                      *   
      *-----------------------------------------------------------------*   
           IF WS-ESQL-FL-THREAD = '1'   
                EXEC SQL ROLLBACK WORK END-EXEC 
    
      *         EXEC SQL END THREAD    END-EXEC 
      *         MOVE '0' TO WS-ESQL-FL-THREAD   
           END-IF   
           .
    
       999-ESQL-GET-ERR.
      *-----------------------------------------------------------------*   
      * DISPLAY DE L'ERREUR ESQL                                        *   
      *-----------------------------------------------------------------*

           MOVE ERROR-STATUS OF RDMCA   
                             TO WS-ESQL-ERROR-STATUS
           MOVE AUX-INFO     OF RDMCA   
                              TO WS-ESQL-AUX-INFO   
           DISPLAY "*** ESQL ERROR ***"          UPON PRINTER   
           DISPLAY "TRACE       = " WS-ESQL-TRACE  UPON PRINTER 
           DISPLAY "SQLCODE     = " SQLCODE      UPON PRINTER   
           DISPLAY "ERROR-SATUS = " ERROR-STATUS OF RDMCA   
                                                 UPON PRINTER   
           DISPLAY "AUX-INFO    = " AUX-INFO     OF RDMCA   
                                                 UPON PRINTER   
           MOVE "0" TO WS-ESQL-FL   
    
           PERFORM UNTIL WS-ESQL-FL = "1"   
                EXEC SQL GETERROR INTO  
                  :WS-ESQL-LB-ERR (1), :WS-ESQL-LB-ERR (2), 
                  :WS-ESQL-LB-ERR (3), :WS-ESQL-LB-ERR (4), 
                  :WS-ESQL-LB-ERR (5)   
                END-EXEC
                DISPLAY WS-ESQL-LB-ERR (1) UPON PRINTER 
                DISPLAY WS-ESQL-LB-ERR (2) UPON PRINTER 
                DISPLAY WS-ESQL-LB-ERR (3) UPON PRINTER 
                DISPLAY WS-ESQL-LB-ERR (4) UPON PRINTER 
                DISPLAY WS-ESQL-LB-ERR (5) UPON PRINTER 
                IF  WS-ESQL-LB-ERR (1) = SPACES 
                AND WS-ESQL-LB-ERR (2) = SPACES 
                AND WS-ESQL-LB-ERR (3) = SPACES 
                AND WS-ESQL-LB-ERR (4) = SPACES 
                AND WS-ESQL-LB-ERR (5) = SPACES 
                     MOVE "1" TO WS-ESQL-FL 
                END-IF  
           END-PERFORM  
    
    
           PERFORM VARYING WS-ESQL-IX-CML FROM 1 BY 1   
                     UNTIL WS-ESQL-IX-CML > 200 
                DISPLAY WS-ESQL-CML (WS-ESQL-IX-CML) UPON PRINTER   
           END-PERFORM  
    
    
           MOVE WS-ESQL-SQLCODE      TO SQLCODE 
           MOVE WS-ESQL-ERROR-STATUS TO ERROR-STATUS OF RDMCA   
           MOVE WS-ESQL-AUX-INFO     TO AUX-INFO     OF RDMCA
