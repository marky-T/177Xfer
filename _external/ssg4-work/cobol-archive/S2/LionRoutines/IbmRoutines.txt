routine main(fileid, currentfilename, destination : string optional)
begin_routine
  if destination = '' //dummy file
    if currentfilename <> '' //for input files, filename is still empty!
      amt_fco.filedelete (currentfilename)
    endif
  elseif length(destination) = 1 //job has assigned this file to a printer

    prout.printer := destination
    prout.filename := si-repname & '__' & today.hhmmssnn & '0.' & fileid &  '.PRN'

    print(printrec)
    freeoutput(prout)

    amt_fco.filemove (currentfilename, prout.filename)
  endif

begin_definitions
const

var
  amt_sqlca ()

booleans

end_definitions

// This routine will convert the si-dbstatus to errorcodes sqlcode and sqlstate.
// The routine gets the SQLCA structure as a var parameter so in the future
// other SQLCA members can be changed here also.
// Error Codes overview:
// execution was successful:                sqlcode = 0, sqlstate = '00000'
// execution was successful with a warning: sqlcode > 0, sqlstate = ?
// execution was not successful:            sqlcode < 0, sqlstate >= '07000'
// no data was found:                       sqlcode = 100, sqlstate >= '02000'

routine main (var par-sqlca : alpha 178)
begin_routine
  sqlca := par-sqlca
  if resok  // Execution was successful
    sqlcode := 0
    sqlstate := '00000'
  else
    startcase si-dbstatus
    case 0  // Unknown error occurred
      sqlcode := -1
      sqlstate := '07000'
      jobsummary ('SQL Error: ' & getlastresult, true)  // log error
    case 1  // Record not found
      sqlcode := 100
      sqlstate := '02000'
    case 2 3 4 5 6  // Error situation, execution was not successful
      sqlcode := -1
      sqlstate := '07000'
      jobsummary ('SQL Error: ' & getlastresult, true)  // log error
    endcase
  endif
  resok := true
  si-dbstatus := 0
  par-sqlca := sqlca
end_routine

routine main (att : alpha 1, b_current : boolean): boolean
begin_routine
  if (att in ['@00@', '@80@', '@D0@']) //preserve current state
    result := b_current
  elseif (att in ['@60@'..'@7F@']) or (att >= '@E0@') or  //these values are ebcdic-based
      (att in ['@2A@', '@2D@', '@2F@', '@59@', '@5A@', '@5F@']) //these values are ascii-based
    result := true
  else
    result := false
  endif
end_routine

begin_definitions
const

var
  hexdigits: alpha 16 value '0123456789ABCDEF'
  a_val: alpha 1 [16] redefines hexdigits
  fnstruct: structure
    i_val: numeric 4 compbinary
  end_structure
  s_out : structure
    a_hex : alpha 1 [4]

routine main
begin_routine
  if eibresp <> dfhresp_normal
    sme ('AMT_CICS ABEND occurred. Respcode: ' & eibresp & ' eibfn: ' & inhex (eibfn), , 'ODT')
    cics_abend ()
  endif

routine inhex (p_val : alpha 2) : alpha 4
begin_routine
  fnstruct := p_val
  i_pos := 4
  loop while i_pos > 0
    i := (i_val mod 16) + 1
    a_hex[i_pos] := a_val[i]
    i_val := i_val / 16
    i_pos -= 1
  endloop
  result := s_out
end_routine

<OBJECT name='AMT_CICS_START' description='CICS functionality'>
<OPTIONS kind='P' init='N'/>
<CODE>

begin_definitions
const

var
  newtask : task
  newjob : job
  startdate : numeric 8
  starttime : numeric 4
  reqid : numeric 8
  //vars used in checktime routine:
  days,
  hours,
  minutes,
  seconds,
  pos      : integer
  integer-time,
  interval : boolean
  time     : numeric 8
  time-r   : structure redefines time
    time-hhmm : structure
      time-hh: numeric 2
      time-mm: numeric 2
    end_structure
    time-ss: numeric 2
    time-nn: numeric 2
  end_structure

booleans

end_definitions


routine main (p_transid: string, p_time: string optional, p_data: realstring optional, p_reqid: string optional, p_options: string optional)
begin_routine
  if si-origin <> '' then eiblk := si-param
  amt_set_eib_error (dfhresp_normal, '@10@@08@')
  si-dbstatus := 1
  if p_reqid <> ''
    //If the request already exists, starting a task is not needed
    qdfhtemp.where (qdfhtemp.requestid = p_reqid)
    qdfhtemp.findfirst (index-dfhtemp)
  endif
  cics-start-options := p_options
  if si-dbstatus = 1
    newtask.jobname := 'STARTTASK' //getprogname (p_transid)
    newtask.appname := si-applicationname
    newtask.connect ()
    if newtask.jobname = ''
      amt_set_eib_error (dfhresp_transiderr, '@10@@08@')
    elseif newtask.connected
      if p_reqid = ''
        reqid := newtask.batchid
        p_reqid := reqid
      endif
      newjob := newtask.createjob ()
      if checktime (p_time)
        newjob.starttime := starttime
        if startdate <> 0 then newjob.startdate := startdate
      endif
      if cics-start-options.termid <> ''
        newjob.adddefaultprinter (cics-start-options.termid)
      endif
      newjob.sysparam := p_reqid
      reqid := newjob.addjobrequest (newtask.jobname, report)
    else
      amt_set_eib_error (dfhresp_invreq, '@10@@08@')
    endif
  endif
  if eibresp = dfhresp_normal

    qdfhtemp.data := p_data
    qdfhtemp.rtransid := cics-start-options.rtransid
    qdfhtemp.rtermid := cics-start-options.rtermid
    qdfhtemp.queue := cics-start-options.queue

routine checktime (p_time: string): boolean
begin_routine
  result := false
  if p_time = '' then exit //return false; have the task executed immediately
  integer-time := false
  interval := false
  if copy(p_time, 1, 1) = '+'
    interval := true
    pos := 2
  else
    pos := 1
  endif
  days := 0
  hours := unstring (p_time, pos, ':')
  if pos > length (p_time) //we have an hhmmss formatted number
    seconds := hours mod 100
    minutes := (hours mod 10000) / 100
    hours := hours / 10000
    if not interval
      integer-time := true
    endif
  else
    minutes := unstring (p_time, pos, ':')
    seconds := unstring (p_time, pos, ':')
  endif
  time := today.hhmmssnn
  if interval
    hours += time-hh
    minutes += time-mm
    seconds += time-ss
  endif
  if seconds > 60
    minutes += seconds / 60
    seconds := seconds mod 60
  endif
  if minutes > 60
    hours += minutes / 60
    minutes := minutes mod 60
  endif
  if hours > 24
    days := hours / 24
    hours := hours mod 24
  endif
  time-hh := hours
  time-mm := minutes
  time-ss := seconds
  if integer-time //if time has been supplied as an exact number in hhmmss format, it must be within the next 18 hours
    if (time < today.hhmmssnn)
      days := 1
      if (time > today.hhmmssnn - 6000000) //time is in the latest 6 hours
        exit //return false; have the task executed immediately
      endif
    elseif (time > today.hhmmssnn + 18000000) //time more than 18 hours forward; supposedly in latest 6 hours
      exit //return false; have the task executed immediately
    endif
  elseif not interval
    if (time < today.hhmmssnn)
      days := 1
    endif
  endif
  result := true
  starttime := time-hhmm
  startdate := 0
  if days > 0
    dateresult.daynum := today.daynum + days
    startdate := dateresult.ccyymmdd
  endif
end_routine

routine main (p_val: integer, p_format: string, p_sep: alpha 1 optional): string
begin_routine
  if p_format = 'TIME'
    p_val := p_val mod 1000000
    result := p_val
    if p_sep <> ''
      result := copy (result, 1, 4) & p_sep & copy (result, 5)
      result := copy (result, 1, 2) & p_sep & copy (result, 3)
    endif
    exit
  endif
  
  dateresult.ccyyddd := p_val / 1000000
  if p_sep <> ''
    startcase p_format
      case 'MMDDYYYY'
        result := dateresult.mm-dd-ccyy
      case 'DDMMYYYY'
        result := dateresult.dd-mm-ccyy
      case 'YYYYMMDD' 'FULLDATE'
        result := dateresult.ccyy-mm-dd
      case 'YYYYDDMM'
        result := dateresult.dd-mm-ccyy
        result := copy (result, 6, 4) & '-' & copy (result, 1, 5)
      case 'YYYYDDD'
        result := dateresult.ccyyddd
        result := copy (result, 1, 4) & '-' & copy (result, 5, 3)
      else //'DATE'
        result := dateresult.yy-mm-dd
    endcase
    if p_sep <> '-'
      loop for i_pos := 1 to length (result)
        if copy (result, i_pos, 1) = '-' then replace (result, i_pos, p_sep)
      endloop
    endif
  else
    startcase p_format
      case 'DATEFORM'
        result := 'YYMMDD'
      case 'DAYCOUNT'
        result := dateresult.daynum
      case 'DAYOFWEEK'
        d_today.ccyymmdd := today.ccyymmdd
        i_shift := dateresult.daynum - d_today.daynum
        result := (today.weekdayno + i_shift - 1) mod 7
        //NOTE: -1 CORRECTION BECAUSE SUNDAY = 1 IN LION, = 0 IN CICS
        if result < 0 then result += 7
      case 'DAYOFMONTH'
        result := dateresult.dd
      case 'MONTHOFYEAR'
        result := dateresult.mm
      case 'YEAR'
        result := dateresult.ccyy
      case 'MMDDYYYY'
        result := dateresult.mmddccyy
      case 'DDMMYYYY'
        result := dateresult.ddmmccyy
      case 'YYYYMMDD' 'FULLDATE'
        result := dateresult.ccyymmdd
      case 'YYYYDDMM'
        result := dateresult.ddmmccyy
        result := copy (result, 5, 4) & copy (result, 1, 4)
      case 'YYYYDDD'
        result := dateresult.ccyyddd
      else //'DATE'
        result := dateresult.yymmdd
    endcase
  endif
end_routine
