begin_definitions
const

var
  daypos : integer value 1
  daynum : alpha 2 value '00'
booleans

end_definitions


routine main (intime string, informat string, units integer, unitformat alpha 1, outformat: realstring optional): realstring
var
    i_time   : integer
    s_format : string
begin_routine
    if outformat = '' then outformat := informat
    if intime = 'TODAY'
      s_format := ''
    else
      s_format := informat
    endif
    if unitformat in ['M', 'Y'] //preserve day number, even if it's invalid
      asy-string := unstring (informat, daypos, 'DD')
      daypos -= 2
      if copy(informat, daypos, 2) = 'DD'
        daynum := copy (intime, daypos, 2)
        if daynum > '28'
          replace (intime, daypos, '01')
        endif
      endif
    endif
    i_time := xtime (intime, s_format, units, unitformat)
    result := xtimeformat (i_time, outformat)
    if daynum > '28'
      if outformat <> informat
        daypos := 1
        asy-string := unstring (outformat, daypos, 'DD')
        daypos -= 2
      endif
      if copy(outformat, daypos, 2) = 'DD'
        replace (result, daypos, daynum)
      endif
    endif
end_routine

<OBJECT name='ASY-SENDTOPRINTER' description='Process COMS output string as print data'>
<OPTIONS kind='P' init='N'/>
<CODE>

begin_definitions
const
    cr      : alpha 1 value '@0D@'
    crlf    : alpha 2 value '@0D@@0A@'
    lf      : alpha 1 value '@0A@'
    ws-null : structure
      asy-filler-1 : numeric 2 comp value 0
    end_structure
    esc            : alpha 1 value '@1B@'
    finaltag       : alpha 2 value '@1b@E'
    reclen         : integer value 96 // 1 added for CR each unstring
var
    len            : integer
    partlen        : integer
    pos-cr         : integer
    pos-crlf       : integer
    prevpos        : integer
    countcr        : integer
    countlf        : integer
    printbuffer    : list(alpha 2000)
    ix_buffer      : integer
    save_msgrecord : alpha 2000
    msgpart        : realstring
    leftover       : realstring

    null_count     : integer
    ff_count       : integer
    lf_count       : integer
    firstline      : boolean
    b_escape       : boolean
    prev_escape    : boolean
booleans

end_definitions

routine main (s_msg : realstring, s_dest : string)
begin_routine
    null_count := 0
    inspect (s_msg)
        counting (null_count, trailing, ws-null)
    endinspect
    if null_count > 0
        s_msg := copy (s_msg, 1, length(s_msg) - null_count)
    endif
    if (s_msg = finaltag)
        if (leftover <> '')
          printrecord (leftover)
        endif
        freeoutput ()
        prout.printer := ''
        exit
    endif
    if (prout.printer = '')
        if (leftover <> '')
          s_msg := leftover & s_msg
          leftover := ''
        endif
        prout.maxlines       := 0
        prout.hookparameters := 'NOTRANS'
        prout.printer        := s_dest
    endif
    b_escape  := (copy(s_msg, 1, 1) = esc)
    firstline := true
    pos-crlf    := 1
    len       := length (s_msg)
    pos-cr    := 1
    loop
        prevpos   := pos-cr
        msgpart := unstring (s_msg, pos-cr, cr, false)
        //Check if the line contains a FF
        //The last FF should not be printed, because this would mean an empty page
        lf_count  := 0
        ff_count  := 0
        inspect (msgpart)
            counting (lf_count, all, lf)
            counting (ff_count, all, '')
            replacing (all, ws-null, ' ')
        endinspect

        if length (msgpart) = lf_count + ff_count
            printbuffer.add(msgpart)
        //This is to prevent a last empty line is being printed
        //PO50 always printed a 2nd page because of an empty line at the end
        else
            if (prevpos + length(msgpart) > len)
                leftover := msgpart
                break
            endif
            if printbuffer.count > 0
                save_msgrecord := msgpart
                loop for ix_buffer := 1 to printbuffer.count
                    printrecord (printbuffer[ix_buffer])
                endloop
                printbuffer.clear ()
                msgpart := save_msgrecord
            endif
            printrecord (msgpart)
        endif
        if pos-cr > len then break
    endloop
    prev_escape := b_escape
end_routine


routine printrecord (printstring: string)
begin_routine
    if copy(printstring, 1, 1) = ''
        if prout.linesprinted > 0 then skip(top)
        printstring := copy(printstring, 2)
    endif
    lf_count := 0
    inspect(printstring)
        counting (lf_count, leading, lf)
    endinspect
    if lf_count > 0
        skip(lf_count - 1)
        printstring := copy(printstring, lf_count + 1)
    endif
    msgrec := printstring
    if (lf_count > 0) or (firstline and (not b_escape) and (not prev_escape) )
        print (msgrec)
    else
        print (msgrec, noskip)
    endif
    firstline := false
end_routine

</CODE>
</OBJECT>

  result := s_tmp

<OBJECT name='CALC-SCROLL-LINE' description='XGEN function'>
<OPTIONS kind='P' init='Y'/>
<CODE>


begin_definitions
const

var
  a_action   : alpha 1
  a_char     : alpha 1
  i_char     : integer
  i_scroll   : integer
  new_page   : integer
booleans

end_definitions


routine main (a_input: alpha 5, i_pagesize: integer): alpha 5
begin_routine
  if (scroll-line = 0)
    scroll-line := 1
    scroll-page := 1
  endif
  //determine action
  loop for i_char := 1 to 5
    a_char := copy(a_input, i_char, 1)
    if a_char in ['+', '-', '>', '<', '@']
      a_action := a_char
      replace (a_input, i_char, ' ')
      break

    elseif a_input in ['L', '.'] //advance to last duplicate
      scroll-line := 900000 //page is calculated in program
      result := '-'
      exit
    endif
  endloop
  //determine number
  a_input := trim(a_input)
  i_scroll := strtoint (a_input)
  if a_input = '' //presume 1
    i_scroll := 1
  endif

  //compute new scroll line depending on action and number
  new_page := 0
  startcase a_action
  case ' ' //absolute page
    result := '+'
    new_page := i_scroll
  case '+' //advance number of pages
    result := '+'
    new_page := scroll-page + i_scroll
  case '-' //backup  number of pages
    result := '-'
    new_page := scroll-page - i_scroll
    scroll-line := 0 //enforce new calculation if new_page <= 0
  case '>' //advance number of entries
    result := '>' & a_input
    scroll-line += i_scroll
  case '<' //backup  number of entries
    result := '<' & a_input
    scroll-line -= i_scroll
  case '@' //absolute entry
    result := '+'
    scroll-line := i_scroll
  endcase

  if new_page > 0
    scroll-line := (new_page - 1) * i_pagesize + 1
    scroll-page := new_page
  elseif scroll-line <= 1
    scroll-line := 1
    scroll-page := 1
    result := '+'
  else
    calc-scroll-page (i_pagesize, scroll-line) //scroll-page is calculated
  endif
end_routine


</CODE>
</OBJECT>

begin_definitions
const

var
  inarr  : numeric 1 [17]
  instr  : string
  runlth : numeric 2 value 18
  runamt : numeric 17.2
  outstr : string
  //vars used in spell999
  teenarray : alpha 10 [19]
  tenarray  : alpha 10 [9]
  intot     : numeric 2

booleans

routine main (inamt financial 17.2, no-zero-cents boolean optional): string
begin_routine
  teenarray[1]  := 'One'
  teenarray[2]  := 'Two'
  teenarray[3]  := 'Three'
  teenarray[4]  := 'Four'
  teenarray[5]  := 'Five'
  teenarray[6]  := 'Six'
  teenarray[7]  := 'Seven'
  teenarray[8]  := 'Eight'
  teenarray[9]  := 'Nine'
  teenarray[10] := 'Ten'
  teenarray[11] := 'Eleven'
  teenarray[12] := 'Twelve'
  teenarray[13] := 'Thirteen'
  teenarray[14] := 'Fourteen'
  teenarray[15] := 'Fifteen'
  teenarray[16] := 'Sixteen'
  teenarray[17] := 'Seventeen'
  teenarray[18] := 'Eighteen'
  teenarray[19] := 'Ninteen'
  
  tenarray[2]   := 'Twenty'
  tenarray[3]   := 'Thirty'
  tenarray[4]   := 'Forty'
  tenarray[5]   := 'Fifty'
  tenarray[6]   := 'Sixty'
  tenarray[7]   := 'Seventy'
  tenarray[8]   := 'Eighty'
  tenarray[9]   := 'Ninety'

  inarr[] := inamt
  instr   := format(inamt)
  runlth  := length(instr)
  runamt  := inamt

  if runlth > 15
      outstr := spell999(inarr[1], inarr[2], inarr[3]) + ' Trillion'
      RUNLTH := RUNLTH -3
  endif
  if runlth > 12
      if inarr[4] + inarr[5] + inarr[6] <> 0
          if outstr <> '' then outstr := outstr + ', '
          outstr := outstr + spell999(inarr[4], inarr[5], inarr[6]) + ' Billion'
      endif
      runlth := runlth -3
  endif
  if runlth > 9
      if inarr[7] + inarr[8] + inarr[9] <> 0
          if outstr <> '' then outstr := outstr + ', '
          outstr := outstr + spell999(inarr[7], inarr[8], inarr[9]) + ' Million'
      endif
      runlth := runlth -3
  endif
  if runlth > 6
      if inarr[10] + inarr[11] + inarr[12] <> 0
          if outstr <> '' then outstr := outstr + ', '
          outstr := outstr + spell999(inarr[10], inarr[11], inarr[12]) + ' Thousand'
      endif
      runlth := runlth -3
  endif
  if runlth > 3
      if inarr[13] + inarr[14] + inarr[15] <> 0
          if outstr <> '' then outstr := outstr + ', '
          outstr := outstr + spell999(inarr[13], inarr[14], inarr[15])
      endif
      runlth := runlth -3
  endif
  if runlth <= 3
      if outstr <> '' then outstr := outstr + ', AND'
      if inarr[16] + inarr[17] = 0 and no-zero-cents = false then outstr := outstr + ' XX/100'
      if inarr[16] + inarr[17] <> 0 then outstr := outstr + ' ' + inarr[16] + '' + inarr[17] + '/100'
  endif
  result := suc(trim(outstr))
end_routine

routine spell999 (inhun numeric 1, inten numeric 1, inone numeric 1): string
begin_routine  
  intot         := inten + '' + inone
  
  if intot = 0
      result := ''
  elseif intot < 20
      result := teenarray[intot]
  else
      result := tenarray[inten] + '-' + teenarray[inone]
  endif
  
  if inhun > 0 and intot <> 0 then  result := teenarray[inhun] + ' Hundred and ' + result
  if inhun > 0 and intot = 0 then  result := teenarray[inhun] + ' Hundred ' + result
  
  result := ' ' + result
end_routine

<OBJECT name='STRTONUM' description='XGEN function'>
<OPTIONS kind='P' init='Y'/>
<CODE>

begin_definitions
const

var
  temp   : string
  invoer : structure
    voor : structure
      voorpos : alpha 1 [18]
    end_structure
    achter : structure
      achterpos : alpha 1 [18]
    end_structure
  end_structure
  uitvoer     : numeric 36.18 redefines invoer
  pos         : integer
  pos1        : integer
  sign        : signed 1 value 1
  signfound   : boolean value false
  numberfound : boolean
  thousandcnt : integer

booleans

end_definitions


routine main (inp : string ) : signed 36.18
begin_routine
  setto('0',invoer)
  resok := true
  pos1  := 1
  pos   := 1
  temp  := unstring (inp,pos1,'.')
  //replace leading and trailing spaces by zeroes
  replace(voor,19 - length(temp),temp)
  // sign trailing no decimals
  if pos1 <= length(inp)
      achter := copy(inp,pos1)
  // end replace leading and trailing spaces by zeroes
  elseif voorpos[18] = '-'
      sign      := -1
      signfound := true
      setto('0',voor)
      temp := unstring(voor,pos,'')
      setto('0',voor)
      replace(voor,19 - length(temp),temp)
  elseif voorpos[18] = '+'
      sign      := +1
      signfound := true
      temp      := unstring(voor,pos,'')
      setto('0',voor)
      replace(voor,19 - length(temp),temp)
  endif
  //end sign trailing no decimals
      
  thousandcnt := 0
  temp        := ''
  loop for pos := 1 to 18
      if numeric(voorpos[pos])
          if voorpos[pos] > '0' then numberfound := true  // only numeric and , allowed
          temp += voorpos[pos]
          if thousandcnt > 0 then thousandcnt += 1
      elseif numberfound
          if voorpos[pos] = ','
              if not thousandcnt in [0,1,4]   // after a thousand seperator three numerics are needed
                  resok  := false
                  result := 0
                  exit module
              endif
              temp        := '0' & temp
              thousandcnt := 1
          else
              resok  := false
              result := 0
              exit module
          endif
      elseif voorpos[pos] = ''   // could be sign is separated by space(s)
          temp += '0'
      elseif voorpos[pos] = '+'
          if signfound
              resok  := false
              result := 0
              exit module
          endif
          temp      := '0' & temp
          sign      := 1
          signfound := true
      elseif voorpos[pos] = '-'
          if signfound
              resok  := false
              result := 0
              exit module
          endif
          temp      := '0' & temp
          sign      := -1
          signfound := true
      else
          resok  := false
          result := 0
          exit module
      endif
  endloop
  if not thousandcnt in [0,1,4]
      resok  := false
      result := 0
      exit module
  endif
  voor        := temp
  numberfound := false
  loop for pos := length(inp) - (pos1 - 1) downto 1
      if numeric(achterpos[pos])
          numberfound := true
      elseif numberfound
          resok  := false
          result := 0
          exit module
      elseif achterpos[pos] = ''  // could be trailing sign is separated by space(s)
          achterpos[pos] := '0'
      elseif achterpos[pos] = '+'
          if signfound
              resok  := false
              result := 0
              exit module
          endif
          achterpos[pos] := '0'
          sign           := 1
          signfound      := true
      elseif achterpos[pos] = '-'
          if signfound
              resok  := false
              result := 0
              exit module
          endif
          achterpos[pos] := '0'
          sign           := -1
          signfound      := true
      else
          resok  := false
          result := 0
          exit module
      endif
  endloop
  result := uitvoer * sign
end_routine

</OBJECT>

<OBJECT name='TESTNUMERICFMT' description='XGEN function'>
<OPTIONS kind='P' init='Y'/>
<CODE>

begin_definitions
const

var
    ch_sign : alpha 1 value ' '
    n_dec   : numeric 18.18
    a_dec   : alpha 18 value '000000000000000000'
    ar_num  : alpha 1[36]
    i_cnt,
    i_pos,
    i_len   : integer
    s_num,
    s_dec   : string
    b_lead,
    b_sep   : boolean

booleans

end_definitions
routine main (s_in: string, i_dig: integer optional, i_dec: integer optional): boolean
begin_routine
    result        := false
    z-ns-results  := 1
    z-ns-dec-found := 0
    asy-auxnumfmt := 0
    //z-ns-dec-format := 0
    if s_in = '' //empty input is interpreted as zero
        z-ns-results  := 0
        result := true
        exit
    endif
    if i_dig = 0 and i_dec = 0
        i_dig := 18
        i_dec := 18
    endif
    ar_num[] := trim (s_in)
    i_len    := length (ar_num[])
    //first, strip the sign off
    if ar_num[1] in ['+', '-']
        ch_sign  := ar_num[1]
        ar_num[] := trim (copy (ar_num[], 2))
    elseif ar_num[i_len] in ['+', '-']
        ch_sign  := ar_num[i_len]
        ar_num[] := trim (copy (ar_num[], 1, i_len - 1))
    endif
    if ch_sign <> ' ' then i_len := length(ar_num[])
    loop for i_pos := 1 to i_len
        if s_num = '' and ar_num[i_pos] = '0'
            //ignore leading zeroes
            b_lead := true
        elseif ar_num[i_pos] = '.' //decimal point
            if b_sep and i_cnt <> 3 then exit
            s_dec := trim(copy (ar_num[], i_pos + 1))
            break
        elseif ar_num[i_pos] = ',' //separator char
            if not b_sep
                if not (i_cnt in [1..3]) then exit //the first group may have 1, 2 or 3 digits (leading zeroes excluded)
                b_sep := true
            else  //after the first separator, we expect groups of 3 digits
                if i_cnt <> 3 then exit
            endif
            b_sep := true
            i_cnt := 0
        elseif ar_num[i_pos] in ['0'..'9']
            i_cnt += 1
            s_num &= ar_num[i_pos]
        elseif i_pos = i_len
            if ar_num[i_pos] in ['{','A'..'I']
                i_cnt   += 1
                s_num   &= overpunchvalue(ar_num[i_pos])
                ch_sign := '+'
            elseif ar_num[i_pos] in ['}','J'..'R']
                i_cnt   += 1
                s_num   &= overpunchvalue(ar_num[i_pos])
                ch_sign := '-'
            else
                exit
            endif
        else
            exit
        endif
    endloop
    if s_num = '' and s_dec = '' and not b_lead then exit
    if s_dec <> ''
        z-ns-dec-found := length (s_dec)
        if not numeric (s_dec) or z-ns-dec-found > i_dec then exit
        replace(a_dec, 1, s_dec)
        n_dec := a_dec
    elseif b_sep and i_cnt <> 3
        exit
    endif
    if s_num <> ''
        if not numeric(s_num) or length (s_num) > i_dig then exit
        asy-auxnumfmt := strtoint (s_num)
    endif
    asy-auxnumfmt += n_dec
    if ch_sign = '-' then asy-auxnumfmt := asy-auxnumfmt * (-1)
    z-ns-results := 0
    result := true
end_routine

    case '{' '}'

    case 'A' 'J'

    case 'B' 'K'
        result := '2'
    case 'C' 'L'
        result := '3'
    case 'D' 'M'
        result := '4'
    case 'E' 'N'
        result := '5'
    case 'F' 'O'
        result := '6'
    case 'G' 'P'

        exit
    elseif s_time = 'TODAY'
        //provide CCYYMMDD and/or ZZUUSSTT, but omit missing lower-order parts or set them to zero
        //if higher-order parts haven't been specified, do supply them
        s_ccyymmdd := today.ccyymmdd
        if pos ('JJJ', s_format) = 0
            if pos('DD', s_format) = 0
                n_dd := 0
                if pos('MM', s_format) = 0
                    n_mm := 0
                    if pos('YY', s_format) = 0 then n_date := 0
                endif
            endif
        endif
        if pos('HH', s_format) > 0 or pos('ZZ', s_format) > 0
            s_time := today.hhmmssnn
            if pos('TT', s_format) > 0
            //ok
            elseif pos('SS', s_format) > 0
                s_time := copy (s_time, 1, 6)
            elseif pos('UU', s_format) > 0
                s_time := copy (s_time, 1, 4)
            else
                s_time := copy (s_time, 1, 2)
            endif
            s_format := copy ('ZZUUSSTT', 1, length (s_time))
            if n_date > 0 //don't supply date if not specified so
                s_time   := s_ccyymmdd + s_time
                s_format := 'CCYYMMDD' + s_format
            endif
        else //don't supply time if not specified so (for sake of comparing dates)
            s_time   := s_ccyymmdd
            s_format := 'CCYYMMDD'
        endif
    endif //TODAY

    i_formatlen := length (s_format)
    if (copy (s_format, i_formatlen, 1) = 'B') //remove trailing Bs from format for sake of length test
        loop while (i_formatlen > 1)
            i_formatlen -= 1
            if (copy (s_format, i_formatlen, 1) <> 'B') then break
        endloop
        s_format := copy (s_format, 1, i_formatlen)
    endif

    if numeric(s_time) and length (s_time) < i_formatlen
        //a number may have been passed with inttostr, so that leading zeroes have been stripped off
        s_time := copy ('0000000000000000', 1, i_formatlen - length (s_time)) & s_time
    endif
    if numeric(s_time) and length (s_time) > i_formatlen
        s_time := copy (s_time, 1 + length (s_time) - i_formatlen, i_formatlen)
    endif

    result := 0
    i_pos  := 1
    loop while i_pos < length (s_format)
        if copy (s_format, i_pos, 7) = 'CCYYJJJ' or copy (s_format, i_pos, 7) = 'CCYYDDD'
            dateresult.ccyyddd := copy (s_time, i_pos, 7)
            if not resok
                result := 0
                exit
            endif
            s_ccyymmdd   := dateresult.ccyymmdd
            i_pos        += 7
            b_ccinformat := true
            b_yyinformat := true
        elseif copy (s_format, i_pos, 5) = 'YYJJJ' or copy (s_format, i_pos, 5) = 'YYDDD'
            dateresult.yyddd := copy (s_time, i_pos, 5)
            if not resok
                result := 0
                exit
            endif
            s_yymmdd     := dateresult.yymmdd
            i_pos        += 5
            b_yyinformat := true
        elseif copy (s_format, i_pos, 1) = '\'
            i_pos += 1
        else
            s_currformat := copy(s_format, i_pos, 2)
            sc s_currformat
            cs 'ZZ'
                i_time          := copy (s_time, i_pos, 2)
                timestruct.i_hh += i_time
                i_pos           += 2
            cs 'HH'
                i_time          := copy (s_time, i_pos, 2)
                timestruct.i_hh := i_time mod 12
                i_pos           += 2
            cs 'UU'
                i_time          := copy (s_time, i_pos, 2)
                timestruct.i_mm := i_time
                i_pos           += 2
            cs 'SS'
                i_time          := copy (s_time, i_pos, 2)
                timestruct.i_ss := i_time
                i_pos           += 2
            cs 'TT'
                i_time          := copy (s_time, i_pos, 2)
                timestruct.i_nn := i_time
                i_pos           += 2
            cs 'AP'
                if copy (s_time, i_pos, 2) = 'PM' then timestruct.i_hh += 12
                i_pos += 2
            cs 'CC'
                n_cc         := copy (s_time, i_pos, 2)
                i_pos        += 2
                b_ccinformat := true
            cs 'YY'
                n_yy         := copy (s_time, i_pos, 2)
                i_pos        += 2
                b_yyinformat := true
            cs 'MM'
                n_mm         := copy (s_time, i_pos, 2)
                i_pos        += 2
                b_mminformat := true
            cs 'DD'
                n_dd         := copy (s_time, i_pos, 2)
                i_pos        += 2
                b_ddinformat := true
            else
                i_pos += 1
            ec
        endif
    endloop

    if n_yy = 0 and not b_yyinformat
        n_yy := today.year // yy needn't be present in input
    endif
    if n_cc = 0 and not b_ccinformat
        if n_yy > gn-century-cutoff
            n_cc := 19
        else
            n_cc := 20
        endif
    endif
    if n_mm = 0 and not b_mminformat
        n_mm := 1 //mm needn't be present in input
    endif
    if n_dd = 0 and not b_ddinformat
        n_dd := 15 //dd needn't be present in input
    endif
    if (n_cc > 21) or (n_cc = 21 and n_yy > 70)
        n_cc := 21    // maximum number of days in numeric 5 (dateresult.numdays) reached
        n_yy := 70
    endif
    dateresult.ccyymmdd := s_ccyymmdd
    if not resok
        result := 0
        exit
    endif
    i_day := dateresult.daynum
    if dateresult.ccyymmdd = today.ccyymmdd
        //today's daynum may be the result of erroneous input (lion complies to linc)
        if dateresult.ccyymmdd <> s_ccyymmdd then i_day := -1
    endif

    if n_time <> 0
        i_time := i_nn + (i_ss * ci_secondfactor) + (i_mm * ci_minutefactor) + (i_hh * ci_hourfactor)
    endif
    if i_addunits <> 0
        addunits(i_addunits, a_units)
    endif
    dateresult.ccyymmdd := n_date
    if not resok
        result := 0
        exit
    endif
    i_day  := dateresult.daynum
    result := (i_day * ci_dayfactor) + i_time
end_routine

routine addunits(i_units: integer, a_unittype: alpha 1)
begin_routine
    startcase a_unittype
    case 'C'
        n_cc += i_units
    case 'Y'
        n_ccyy += i_units
    case 'M'
        i_units += n_mm + (n_ccyy * 12)
        n_mm    := i_units mod 12
        n_ccyy  := i_units / 12

        if n_mm = 0
            n_mm   := 12
            n_ccyy -= 1
        endif
    case 'D'
        dateresult.ccyymmdd := n_date
        dateresult.daynum   := dateresult.daynum + i_units
        n_date              := dateresult.ccyymmdd
    case 'H' 'Z'
        i_time += i_units * ci_hourfactor
    case 'U'
        i_time += i_units * ci_minutefactor
    case 'S'
        i_time += i_units * ci_secondfactor
    case 'T'
        i_time += i_units
    endcase
end_routine

routine pos (s_substr: realstring, s_entire: realstring): integer
begin_routine
    //return the start pos of the substring in the entire string
    //return zero if not found
    i_slen := length (s_substr)
    i_max  := length (s_entire) - i_slen + 1
    result := 1
    loop while result <= i_max
        if copy(s_entire, result, i_slen) = s_substr then exit
        result += 1
    endloop
    result := 0 //not found
end_routine
