@LAB210: MSG 210. LISTE DES DEPENSES PAR SINISTRE - ytd. RSA
@ .      ===================================================
@ . Liste des dépenses entre 1/1 et fin du mois écoulé  
@ADD       SYS$*RUNS.REPRISE
LAB210  
@ . 
@SETC      0/S2 
@DELETE,C  SCTD01*IMP-RSA.  
@ASG,UP    SCTD01*IMP-RSA.,F///4000 
@XQT,E SYS$LIB$*RSA.RSAC-COIVP  
BEGIN THREAD FOR UDSSRC READ;   
USE DEFAULT QUALIFIER AFLINC;   
UNLOAD TO FILE "SCTD01*IMP-RSA" 
    
SELECT no_sin, sum(md_cpt)  
  FROM SCD201_OPR_FNC     SOF,  
       SCD202_LGN_MON     SLM,  
       SCD108_CAT_CPT_COO SCCC  
 WHERE SOF.CD_ETA         = 'DEBI'  
   AND SOF.DT_ETA_TRI between year(curdate())*10000 
                      and     year(curdate())*10000+month(curdate())*100
   AND SOF.FL_CPN_GEN     = 'Y' 
   AND SLM.NO_OPR_FNC     = SOF.NO_OPR_FNC  
   AND SLM.CD_CAT_CPT     = SCCC.CD_CAT_CPT 
   AND SCCC.DT_DEB       <= SOF.DT_CPT  
   AND SCCC.DT_FIN       >= SOF.DT_CPT  
   AND SCCC.FL_CUT        = 'Y' 
 GROUP BY no_sin
 ORDER BY no_sin
    
FORMAT EXTERNAL;
EXIT;   
@FREE      SCTD01*IMP-RSA.  
@TEST      TE/0/S2  
@JUMP      ERRC 
@JUMP      OK   
@ERRC: MSG ** PROBLEME DANS L'UNLOAD ** 
@ABORT  
@OK:   MSG ** UNLOAD OK **  
@TIME
