DEBUT:  
MODE LINE   
ACCEPT PROMPT="ENTRER LE NOM DU FICHIER TEST : " VARIABLE=%FILE 
ACCEPT PROMPT="ENTRER LE NOM DU PROGRAMME    : " VARIABLE=%NAME 
OLD  FILE=%FILE&"."&%NAME   
IF $COMMANDERROR==TRUE  
   DISP "*** ERREUR FICHIER : "&%FILE&"."&%NAME 
   GOTO DEBUT   
ENDIF   
FREE FILE=%FILE 
RENU START=1 INCREMENT=1
    
RECH:   
L CHABAN
IF $COMMANDERROR==TRUE  
   GOTO FINI
ENDIF   
SET %REC:=$C
C /CHABAN/CHAB100/  
L OUTPUT
SET %W1:=$SEARCH($TEXT($C),",") 
SET %W2:=$SEARCH($TEXT($C),",",%W1+1)   
SET %W3:=%W2-%W1-1  
SET %TX:=$SUBSTRING($TEXT($C),%W1+1,%W3)
SET %NB:=$PAD(%TX,5,LEFT,"0")   
SET %NUM:=%NB/100   
IF $FRACTION(%NUM)>0
   SET %NUM:=%NUM+1 
ENDIF   
SET %NUM:=$INTEGER(%NUM)
L "@X"  
GO +1   
SET %D:=$C  
L "@"   
SET %F:=$C-1
GO %REC 
L NUMB  
SET %V:=$SEARCH($TEXT($C),",")  
SET %L:=%V-5
SET %T:=$SUBSTRING($TEXT($C),5,%L)  
SET %N:=$NUMBER(%T) 
C (%N,%NUM) 
    
BOUCLE: 
L NUMB RANGE=$C+1:%D
IF $COMMANDERROR==TRUE  
   GOTO SUITE   
ENDIF   
C (%N,%NUM) 
GOTO BOUCLE 
    
SUITE:  
GENERATE VALUE="*** "&%NB  START=%D-1   

BCL:
IF $C>=%F   
   GOTO FBCL
ENDIF   
GO +%N  
INSERT ***  
GOTO BCL
    
FBCL:   
CREATE FILE=FI. LIFE=TEMPORARY  
MERGE TO=FI.   RANGE=%D-1:%F+1 START=1 INCREMENT=1  
DELETE RANGE=%D-1:%F+1 OUTPUT=BRIEF 
CREATE FILE=FO. LIFE=TEMPORARY  
XQT FILE=IPF*UTILIT.TEST END=YES
MERGE FROM=FO.   RANGE=ALL   START=%D INCREMENT=1   
FREE FILE=FI.   
FREE FILE=FO.   
GOTO RECH   
    
FINI:   
MODE SCREEN 
DISP "*** END PERFOR ***"
