        SUBROUTINE STATIS(ITAB,IFIN)
        DIMENSION ITAB(1000)  
        IFIN=IFIN*3     
        WRITE (6,200)   
        WRITE (6,100) (ITAB(I),I=1,IFIN)  
  100   FORMAT (5(2X,A6,A6,1X,I9),/)
  200  FORMAT (1H1,10X,38HUTILISATION DES DIFFERENTS PARAGRAPHES,//)    
        RETURN    
        END
